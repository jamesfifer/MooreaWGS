#generate 100 different vcfs of 15% of 231 (34) and 85% (197)
#NOTE: sometimes this script acts weird if the directory already exists, recommend deleting existing directory before running/specifying new output_directory
#!/bin/bash
input_file="MayOct2bamscl4vcf"
output_directory="MayOct2PGS_holdout"

# Create the output directory if it doesn't exist
rm -rf "$output_directory"
mkdir -p "$output_directory"

# Count the total number of lines in the input file
total_lines=$(wc -l < "$input_file")

# Loop 100 times
for ((i=1; i<=100; i++)); do
  # Generate a random sample of 34 line numbers and convert it into an array
  sampled_lines=($(shuf -i 1-"$total_lines" -n 34 | sort -n))

  # Create an array of all line numbers
  all_lines=($(seq 1 "$total_lines"))

  # Create the remaining_lines array by filtering elements not present in sampled_lines
  remaining_lines=()
  for element in "${all_lines[@]}"; do
    if [[ ! " ${sampled_lines[*]} " =~ " $element " ]]; then
      remaining_lines+=("$element")
    fi
  done

  # Create the output file names
  output_file1="$output_directory/${input_file}.holdout.REP.$i"
  output_file2="$output_directory/${input_file}.train.REP.$i"

  # Create the output files
  touch "$output_file1"
  touch "$output_file2"

  # Loop through the line numbers and extract the corresponding lines from the input file
  for line_number in ${sampled_lines[@]}; do
    sed -n "${line_number}p" "$input_file" >> "$output_file1"
  done

  for line_number in ${remaining_lines[@]}; do
    sed -n "${line_number}p" "$input_file" >> "$output_file2"
  done
done
